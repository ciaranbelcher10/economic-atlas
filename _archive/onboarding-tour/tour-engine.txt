  // Highlights whichever real, suggested item(s) currently match a
  // selector inside a picker -- re-applied on every DOM mutation inside
  // the container, since these items get destroyed and rebuilt on every
  // selection (confirmed live: a plain one-time class-add loses the
  // highlight the moment the picker re-renders for any reason). Matching
  // by the item's own data-country/data-key attributes (set fresh on
  // every rebuild) rather than holding a node reference is what makes
  // this survive re-renders.
  function applySuggestHighlight(step){
    clearSuggestHighlight();
    if(!step.suggestHighlight) return;
    var containerEl = document.querySelector(step.suggestHighlight.container);
    if(!containerEl) return;
    var sel = step.suggestHighlight.selector;
    function refresh(){
      suggestEls.forEach(function(el){ el.classList.remove("tour-highlight"); });
      suggestEls = Array.prototype.slice.call(document.querySelectorAll(sel));
      suggestEls.forEach(function(el){ el.classList.add("tour-highlight"); });
    }
    refresh();
    if(window.MutationObserver){
      suggestObserver = new MutationObserver(refresh);
      suggestObserver.observe(containerEl, { childList: true, subtree: true });
    }
  }

  function removeBox(){

    // leaving an orphaned box on screen. removeBox() already runs on
    // every single transition (forward, back, exit), so cleaning up
    // here covers all of them in one place.

  function applySiblingRestriction(step, target){
    clearSiblingRestriction();
    if(!step.restrictSiblings || !target) return;
    var els = document.querySelectorAll(step.restrictSiblings);
    var saved = [];
    els.forEach(function(el){
      if(el !== target){
        saved.push({ el: el, pe: el.style.pointerEvents, op: el.style.opacity });
        el.style.pointerEvents = "none";
        el.style.opacity = "0.4";
      }
    });
    restoreSiblingsFn = function(){
      saved.forEach(function(s){ s.el.style.pointerEvents = s.pe; s.el.style.opacity = s.op; });
    };
  }
  function positionAnchorBox(box, target, done){
    var place = function(){
      var r = target.getBoundingClientRect();
      var boxRect = box.getBoundingClientRect();
      var boxWidth = boxRect.width || 340;
      var boxHeight = boxRect.height || 180;
      var centeredLeft = r.left + (r.width / 2) - (boxWidth / 2);
      var left = Math.min(Math.max(12, centeredLeft), window.innerWidth - boxWidth - 12);
      var top = r.bottom + 12;
      if(top + boxHeight + 12 > window.innerHeight){
        top = r.top - boxHeight - 12;
        if(top < 12) top = 12;
      }
      box.style.left = left + "px";
      box.style.top = top + "px";
      if(done) done();
    };
    if(target.scrollIntoView){
      target.scrollIntoView({ block: "center", inline: "nearest", behavior: "auto" });
      requestAnimationFrame(function(){ requestAnimationFrame(place); });
    } else {
      place();
    }
  }
  function resolveTarget(step){
    if(!step.target) return null;
    var el = (typeof step.target === "function") ? step.target() : document.querySelector(step.target);
    if(!el) return null;
    if(el.offsetWidth === 0 && el.offsetHeight === 0) return null;
    return el;
  }
  function resolveBody(step){
    return (typeof step.body === "function") ? step.body() : step.body;
  }
  // Some anchor targets (e.g. Compare's "Add countries" button) are
  // created dynamically by the page's own JS, only after its own async
  // data has loaded -- this script runs near the end of the document,
  // but that doesn't mean everything on the page is actually ready yet;
  // async fetches don't block synchronous script execution. Retrying
  // briefly before falling back to a centered, unhighlighted box (which
  // is what "the highlight doesn't appear" looked like) covers this
  // without needing to know which specific steps are affected.
  function renderStepWhenReady(index, attempts){
    attempts = attempts || 0;
    var step = TOUR_STEPS[index];
    if(step && step.type === "anchor" && step.page === THIS_PAGE){
      if(!resolveTarget(step) && attempts < 30){ // up to ~3s at 100ms
        setTimeout(function(){ renderStepWhenReady(index, attempts + 1); }, 100);
        return;
      }
    }
    renderStep(index);
  }
  function renderStep(index){
    removeBox();
    var step = TOUR_STEPS[index];
    if(!step || step.page !== THIS_PAGE){
      if(step) console.log("[EATLAS_TOUR] renderStep() bailed -- step belongs to page '" + step.page + "' but THIS_PAGE is '" + THIS_PAGE + "'.");
      return; // this step belongs to a page not yet reached/built
    }
    if(!stepVisible(step)){
      var alt = nextVisibleIndex(index + 1, 1);
      if(alt !== -1) advance(alt); else exitTour();
      return;
    }
    if(step.onEnter) step.onEnter();
    var target = null;
    if(step.type === "anchor") target = resolveTarget(step);
    var effectiveType = (step.type === "anchor" && !target) ? "center" : step.type;
    var box = document.createElement("div");
    box.className = "tour-box" + (effectiveType === "center" ? " tour-center" : "");
    box.style.visibility = "hidden"; // measured before being positioned/shown, to avoid a visible jump
    box.innerHTML =
      "<h4>" + step.title + "</h4>" +
      "<p>" + resolveBody(step) + "</p>" +
      "<div class=\"tour-row\">" +
        "<div class=\"tour-actions\">" +
          (nextVisibleIndexOnPage(index - 1, -1) !== -1 ? "<button type=\"button\" class=\"tour-prev\">Back</button>" : "") +
          "<button type=\"button\" class=\"tour-next\">" + (nextVisibleIndexOnPage(index + 1, 1) === -1 ? "Got it" : "Next") + "</button>" +
        "</div>" +
      "</div>" +
      "<div style=\"margin-top:10px; text-align:right;\"><button type=\"button\" class=\"tour-exit\">Exit tour</button></div>";
    document.body.appendChild(box);
    var advanceToNext = function(){
      var n = nextVisibleIndex(index + 1, 1);
      advance(n === -1 ? TOUR_STEPS.length : n);
    };
    if(effectiveType === "anchor" && target){
      if(!step.suppressTargetHighlight){
        target.classList.add("tour-highlight");
        highlightEl = target;
      }
      positionAnchorBox(box, target, function(){
        box.style.visibility = "";
        requestAnimationFrame(function(){ box.classList.add("show"); });
      });
      applySiblingRestriction(step, target);
      applySuggestHighlight(step);
      // Following the instruction (clicking the real, live element the
      // box is pointing at) advances the tour on its own -- Next stays
      // available too, purely as a fallback. Exception: steps marked
      // exploratory (map zoom/pan, year-scrubbing, adding several
      // countries) deliberately skip this -- a single click firing
      // auto-advance cuts off exploration the moment someone starts,
      // rather than after they're actually done playing with it. Next
      // is the only way forward on these, on purpose.
      if(!step.exploratory){
        activeAdvanceTarget = target;
        activeAdvanceHandler = advanceToNext;
        target.addEventListener("click", advanceToNext, { once: true });
      }
    } else {
      box.style.visibility = "";
      requestAnimationFrame(function(){ box.classList.add("show"); });
    }
    var nextBtn = box.querySelector(".tour-next");
    var prevBtn = box.querySelector(".tour-prev");
    var exitBtn = box.querySelector(".tour-exit");
    if(nextBtn) nextBtn.addEventListener("click", advanceToNext);
    if(prevBtn) prevBtn.addEventListener("click", function(){
      var p = nextVisibleIndex(index - 1, -1);
      if(p !== -1) advance(p);
    });
    if(exitBtn) exitBtn.addEventListener("click", exitTour);
    boxEl = box;
  }
  function advance(newIndex){
    if(newIndex >= TOUR_STEPS.length || newIndex < 0){ exitTour(); return; }
    saveState({ active: true, stepIndex: newIndex });
    var step = TOUR_STEPS[newIndex];
    if(step.page === THIS_PAGE){
      renderStepWhenReady(newIndex);
    } else {
      // Next step lives on a different page than this one (e.g. a map
      // click just navigated away) -- nothing more to render here;
      // that page's own copy of this engine picks up rendering when it
      // loads, using the state just saved above.
      removeBox();
    }
  }
  function exitTour(){
    saveState({ active: false, stepIndex: 0 });
    markSeen();
    removeBox();
  }
  function startTour(){
    markSeen();
    var first = nextVisibleIndex(0, 1);
    if(first === -1){
      saveState({ active: true, stepIndex: 0 });
      return;
    }
    saveState({ active: true, stepIndex: first });
    if(TOUR_STEPS[first].page !== THIS_PAGE){
      // The tour always starts from its first step (currently the
      // index page), but "Take the tour" lives in the footer of every
      // page -- clicking it from anywhere else silently did nothing
      // visible: renderStep() correctly refuses to render a step that
      // belongs to a different page, but nothing was navigating there
      // first (confirmed live). Navigate to the right page first; the
      // resume-on-load check at the bottom of this file picks the
      // saved state back up and shows the tour there.
      window.location.href = TOUR_STEPS[first].page;
      return;
    }
    renderStepWhenReady(first);
  }
  window.EATLAS_TOUR = { start: startTour, exit: exitTour };
  var authModalEl = document.getElementById("authModal");
  if(authModalEl && window.MutationObserver){
    new MutationObserver(function(){
      if(!authModalEl.hidden) removeBox();
    }).observe(authModalEl, { attributes: true, attributeFilter: ["hidden"] });
  }
  var replayBtn = document.getElementById("tourReplayLink");
  if(replayBtn) replayBtn.addEventListener("click", startTour);
  // Tour disabled entirely, for everyone, on every device -- auto-start
  // and resume are both hard-disabled below rather than removed outright,
  // so re-enabling later (if ever wanted) is a one-line change back to
  // the previous width-gated condition rather than rebuilding this block.
  var state = getState();
  if(false){
    if(state.active && TOUR_STEPS[state.stepIndex] && TOUR_STEPS[state.stepIndex].page === THIS_PAGE){
      renderStepWhenReady(state.stepIndex);
    } else if(!state.active && !hasSeenBefore()){
      startTour();
    }
  }
})();
</script>
